§ 02 — Binary Frames

64-Byte Header Structure

All binary TCI frames begin with a fixed 64-byte header followed by float32 sample payload.

HEADER (64 bytes) — big-endian
frame_typeuint8
rx_indexuint8
sample_rateuint32
formatuint8
channelsuint8
sample_countuint32
reserved / padding52 bytes
float32 samples (channels interleaved: L/R or I/Q)
frame_type: 0=IQ, 1=RX audio, 2=TX audio, 3=chrono
format: 0=float32LE (typical)
channels: 1=mono, 2=stereo/IQ

§ 04 — Key Deviations

5 Deviations That Require Client Workarounds

A universal TCI client must handle these behavioral differences across implementations.

DEVIATION 01
IQ Exclusive Ownership
deskHPSDR grants IQ streaming to the first caller only. Subsequent iq_start calls are silently accepted. The owner's iq_stop halts all IQ streams, including those of other clients.
Fix: detect via protocol:deskHPSDR in init block; implement backoff + reconnect on stream timeout.
Thetis/NereusSDR: Audio & IQ Stubs
DEVIATION 02
audio_start, audio_stop, iq_start, iq_stop are accepted with no error but produce no binary frames. NereusSDR provides SETUP: flags to enable streaming.
Fix: send SETUP:TciAlwaysStreamIq=true; for NereusSDR, or use WAV audio engine instead of TCI audio.
PTT Not Broadcast
DEVIATION 03
Thetis and NereusSDR echo trx:0,true; only to the requesting client. Other connected clients receive no PTT state change notification.
Fix: maintain own TX state per connection; don't rely on receiving trx: events from other clients.
AGC Command Name
DEVIATION 04
ExpertSDR3 uses agc_auto_ex:trx,rx,mode,… with numeric mode values. All open-source implementations use agc_mode:trx,rx,NAME with named strings like FAST, SLOW.
Fix: detect ExpertSDR3 via init block, use agc_auto_ex for it and agc_mode for everything else.
Modulation Aliases
DEVIATION 05
deskHPSDR translates CWL/CWU → CW and NFM → FM in both directions. ExpertSDR and Zeus preserve CWL/CWU and use FM (not NFM).
Fix: send both CW and CWL/CWU; accept all variants when parsing incoming modulation events.

connect-and-parse-init.js 
const ws = new WebSocket('ws://localhost:50001');
const state = { ready: false, protocol: null, device: null };

ws.onmessage = (event) => {
  if (typeof event.data === 'string') {
    // Text frame: command:arg1,arg2;
    const text = event.data.trim();

    if (text === 'ready') {
      state.ready = true;
      console.log('TCI server ready:', state);
      onReady(ws, state);
      return;
    }

    const m = text.match(/^(\w+):(.*);?$/);
    if (!m) return;
    const [, cmd, args] = m;
    const argv = args.split(',');

    switch (cmd) {
      case 'protocol': state.protocol = argv[0]; break;
      case 'device':   state.device   = argv[0]; break;
      case 'trx_count': state.trxCount = +argv[0]; break;
      case 'dds':  /* initial DDS freq */ break;
      case 'trx':  onTrxChange(+argv[0], argv[1]==='true'); break;
      case 'vfo':  onVfoChange(+argv[0], +argv[1], +argv[2]); break;
    }
  } else {
    // Binary frame: ArrayBuffer with 64-byte header
    parseBinaryFrame(event.data);
  }
};

parse-iq-frame.js 
function parseBinaryFrame(buffer) {
  const view = new DataView(buffer);
  const frameType  = view.getUint8(0);
  const rxIndex    = view.getUint8(1);
  const sampleRate = view.getUint32(2, false); // BE
  const format     = view.getUint8(6);
  const channels   = view.getUint8(7);
  const sampleCount= view.getUint32(8, false); // BE

  if (frameType === 0) { // IQ frame
    const samples = new Float32Array(
      buffer, 64, sampleCount * channels
    );
    // Even indices = I, odd = Q
    processIQ(rxIndex, sampleRate, samples);
  }
}
rx-audio-flow.js 
// RX audio: server → client (frame_type = 1)

function startRxAudio(ws, state, rx = 0, rate = 8000) {
  if (state.protocol === 'NereusSDR') {
    // NereusSDR needs flag to enable audio streaming
    ws.send('SETUP:TciAlwaysStreamIq=true;');
  }
  ws.send(`audio_start:${rx};`);
  // Note: Thetis/NereusSDR accept but may not stream
  // Zeus supports: audio_start:0,true;  (2-arg form)
}

// TX audio: client → server (frame_type = 2)
function sendTxAudio(ws, float32Samples, sampleRate = 8000) {
  const HEADER_SIZE = 64;
  const buf = new ArrayBuffer(HEADER_SIZE + float32Samples.length * 4);
  const view = new DataView(buf);
  view.setUint8(0, 2);                        // frame_type = TX audio
  view.setUint8(1, 0);                        // rx index
  view.setUint32(2, sampleRate, false);       // BE sample rate
  view.setUint8(6, 0);                        // format = float32
  view.setUint8(7, 1);                        // 1 channel (mono TX)
  view.setUint32(8, float32Samples.length, false); // sample count
  const samples = new Float32Array(buf, HEADER_SIZE);
  samples.set(float32Samples);
  ws.send(buf);
}

// Other connected clients do NOT receive the trx: event
// If your app needs to share PTT state across connections,

// Zeus note: echoes the post-call truth (not the requested value)
// so trx:0,true; may echo trx:0,false; if the radio vetoed TX.

ws.onmessage = (event) => {
  if (typeof event.data !== 'string') return;
  const m = event.data.match(/^trx:(\d+),(true|false)/);
  if (m) {
    const [, trxIdx, isTransmitting] = m;
    updatePttState(+trxIdx, isTransmitting === 'true');

function setMode(ws, state, trx, mode) {
  const isDesk = state.protocol === 'deskHPSDR';
  const m = (isDesk && (mode==='CWL'||mode==='CWU'))
    ? 'CW' : mode;
  ws.send(`modulation:${trx},${m};`);
}
agc-mode.js 
// AGC is the biggest naming deviation.
// ExpertSDR3: agc_auto_ex:trx,rx,attack,decay,hang,gain
// All open-source: agc_mode:trx,rx,NAME

function setAGC(ws, state, trx, rx, mode) {
  if (state.protocol === 'ExpertSDR3') {
    // Named modes → numeric ExpertSDR3 params
    const presets = {
      FAST:   '0.002,0.500,0.250,-20',
      MEDIUM: '0.002,1.200,0.500,-20',
      SLOW:   '0.002,3.000,2.000,-20',
      LONG:   '0.002,5.000,3.000,-20',
      FIXED:  '0.002,0.500,0.250,0',
    };
    const p = presets[mode] || presets.MEDIUM;
    ws.send(`agc_auto_ex:${trx},${rx},${p};`);
  } else {
    // All open-source implementations
    ws.send(`agc_mode:${trx},${rx},${mode};`);
  }
}

server-detect.js 
function detectServer(initLines) {
  const proto  = (initLines.find(l => l.startsWith('protocol:')) || '').split(':')[1]?.split(',')[0];
  const device = (initLines.find(l => l.startsWith('device:'))   || '').split(':')[1]?.replace(';','');

  return {
    isExpertSDR3: proto === 'ExpertSDR3' && device !== 'Zeus',
    isZeus:       proto === 'ExpertSDR3' && device === 'Zeus',
    isDeskHPSDR:  proto === 'deskHPSDR',
    isThetis:     proto === 'Thetis',
    isNereusSDR:  proto === 'NereusSDR',
    useAgcAutoEx: proto === 'ExpertSDR3' && device !== 'Zeus',
    exclusiveIQ:  proto === 'deskHPSDR',
    audioStubs:   proto === 'Thetis' || proto === 'NereusSDR',
    noBroadcastPtt: proto === 'Thetis' || proto === 'NereusSDR',
    cwAliased:    proto === 'deskHPSDR' || proto === 'Thetis',
  };
}
§ 07 — AGC Reference

AGC Mode Values by Implementation

The most common source of inter-operability failures is the AGC command name and value format.

Mode Open-source: agc_mode:0,0,NAME; ExpertSDR3: agc_auto_ex:0,0,…; Notes
FAST FAST attack=0.002, decay=0.500 Most commonly used for SSB
MEDIUM MEDIUM or MED attack=0.002, decay=1.200 NereusSDR → NORMAL alias maps here
SLOW SLOW attack=0.002, decay=3.000 For digital modes, slow signals
LONG LONG attack=0.002, decay=5.000 Slowest response, broadcast AM
FIXED FIXED gain=0 (manual RF gain) Zeus: OFF maps to FIXED
§ 08 — Universal Client

Recipe for a Universal TCI Client

A client that handles all 5 deviations will work with every known TCI implementation.

1
Parse the init block before any command
Wait for ready. Build a server-capabilities object from protocol: and device: lines.
ws.send(...) only after state.ready === true
2
Branch on AGC command
Use agc_auto_ex for ExpertSDR3 (when device ≠ Zeus), agc_mode for everything else.
if (srv.isExpertSDR3) agc_auto_ex else agc_mode
3
Accept both CW alias forms
When parsing incoming modulation: events, normalize CW → CWL or CWU based on last-sent VFO offset. When sending, map CWL/CWU → CW for deskHPSDR.
const m = (isDeskHPSDR && isCW) ? 'CW' : mode;
4
Handle audio/IQ stubs gracefully
For Thetis, fall back to WAV/pipe audio. For NereusSDR, send SETUP:TciAlwaysStreamIq=true; before iq_start. Implement timeout detection: if no binary frame in 2s after iq_start, the server probably stubbed it.
setTimeout(() => { if (!gotFrame) handleStub(); }, 2000);
5
Don't rely on PTT broadcast from other clients
Maintain own TX state. For deskHPSDR, be aware that your iq_stop will terminate IQ streams for all connected clients — call it only when intentional.
// Track: myTxState = false; — never assume trx: event from others
universal-tci-client-skeleton.js 
class UniversalTciClient {
  constructor(url) {
    this.ws = new WebSocket(url);
    this.srv = {}; // server capabilities
    this.ready = false;
    this.initLines = [];
    this.ws.onmessage = (e) => this.handleMessage(e);
  }

  handleMessage({ data }) {
    if (data instanceof ArrayBuffer) {
      this.parseFrame(data); return;
    }
    const line = data.trim();
    if (!this.ready) {
      this.initLines.push(line);
      if (line === 'ready') {
        this.srv = detectServer(this.initLines);
        this.ready = true;
        this.onReady();
      }
    } else {
      this.parseCommand(line);
    }
  }

  setAGC(trx, rx, mode)    { agcCommand(this.ws, this.srv, trx, rx, mode); }
  setMode(trx, mode)        { modeCommand(this.ws, this.srv, trx, mode); }
  startIQ(rx, rate = 48000) { iqStart(this.ws, this.srv, rx, rate); }
  startAudio(rx)            { audioStart(this.ws, this.srv, rx); }
  setPTT(trx, tx)           { this.ws.send(`trx:${trx},${tx};`); }
  setVFO(trx, ch, freq)     { this.ws.send(`vfo:${trx},${ch},${freq};`); }
}
